```cpp
// Pattern.h: one drum pattern, 16 pads x 64 steps of velocity (0 = no hit), plus hex rows for the patch file.
// No Rack dependency.
// Author: Ari Klopfer / A_Development. MIT
#pragma once
#include <algorithm>
#include <cstdint>
#include <string>

constexpr int PADS = 16, MAXSTEPS = 64, PATTERNS = 16;

struct Pattern
{
    uint8_t vel[PADS][MAXSTEPS] = {};
    int length = 16;

    void set (int pad, int step, int v) { vel[pad][step] = (uint8_t) std::max (0, std::min (127, v)); }
    void toggle (int pad, int step) { vel[pad][step] = vel[pad][step] ? 0 : 100; }
    void clear() { for (auto& row : vel) for (auto& s : row) s = 0; }
};

// Two lowercase hex digits per step.
inline std::string rowToHex (const uint8_t* row)
{
    static const char* d = "0123456789abcdef";
    std::string s;
    for (int i = 0; i < MAXSTEPS; ++i) { s += d[row[i] >> 4]; s += d[row[i] & 15]; }
    return s;
}

// Writes row only when the whole string is valid.
inline bool rowFromHex (const std::string& s, uint8_t* row)
{
    if (s.size() != 2 * MAXSTEPS) return false;
    auto h = [] (char c) -> int { return c >= '0' && c <= '9' ? c - '0' : c >= 'a' && c <= 'f' ? c - 'a' + 10 : -1; };
    uint8_t tmp[MAXSTEPS];
    for (int i = 0; i < MAXSTEPS; ++i)
    {
        const int a = h (s[2 * i]), b = h (s[2 * i + 1]);
        if (a < 0 || b < 0) return false;
        tmp[i] = (uint8_t) std::min (a * 16 + b, 127);
    }
    std::copy (tmp, tmp + MAXSTEPS, row);
    return true;
}
```

```cpp
// StepClock.h: sixteenth-note step clock. tick(dt) runs it from BPM; pulse() steps it from an external clock.
// Both return true when a new step starts; step is the index just started. Swing delays odd steps.
// No Rack dependency.
// Author: Ari Klopfer / A_Development. MIT
#pragma once

struct StepClock
{
    int length = 16, step = -1;
    double bpm = 120, swing = 0, t = 0;   // swing 0..0.5: even steps last (1+swing) steps, odd ones (1-swing)
    bool first = true;

    void reset() { step = -1; t = 0; first = true; }

    // ponytail: swing only applies to the internal clock; an external clock keeps its own timing
    bool tick (double dt)
    {
        if (first) { first = false; step = 0; t = 0; return true; }
        t += dt;
        const double d = 15.0 / bpm, dur = (step & 1) ? d * (1 - swing) : d * (1 + swing);
        if (t < dur) return false;
        t -= dur;
        step = (step + 1) % length;
        return true;
    }

    bool pulse()
    {
        first = false;
        step = (step + 1) % length;
        t = 0;
        return true;
    }
};
```

```cpp
// Sample.h: a stereo sample and a WAV parser (16/24-bit PCM, 32-bit float; mono or stereo). No Rack dependency.
// Mono is stored as two equal channels. Little-endian hosts only (x86, ARM).
// Author: Ari Klopfer / A_Development. MIT
#pragma once
#include <algorithm>
#include <cmath>
#include <cstdint>
#include <cstring>
#include <vector>

struct Sample
{
    std::vector<float> l, r;
    float rate = 48000.f;
    int size() const { return (int) l.size(); }
};

namespace wavfile
{
constexpr size_t MAX_FRAMES = 48000 * 30;   // one pad holds at most 30 s; longer files are cut

inline uint32_t u32 (const uint8_t* p) { return p[0] | p[1] << 8 | p[2] << 16 | (uint32_t) p[3] << 24; }
inline uint16_t u16 (const uint8_t* p) { return (uint16_t) (p[0] | p[1] << 8); }

inline bool parse (const uint8_t* d, size_t n, Sample& out)
{
    if (n < 12 || std::memcmp (d, "RIFF", 4) || std::memcmp (d + 8, "WAVE", 4)) return false;
    int tag = 0, ch = 0, bits = 0;
    uint32_t rate = 0;
    for (size_t i = 12; i + 8 <= n;)
    {
        const uint32_t len = u32 (d + i + 4);
        const uint8_t* c = d + i + 8;
        const size_t avail = n - (i + 8), body = len < avail ? len : avail;   // a lying size is clipped to the file
        if (! std::memcmp (d + i, "fmt ", 4) && body >= 16)
        {
            tag = u16 (c); ch = u16 (c + 2); rate = u32 (c + 4); bits = u16 (c + 14);
            if (tag == 0xFFFE && body >= 26) tag = u16 (c + 24);   // WAVE_FORMAT_EXTENSIBLE: real tag is the sub-format's first word
        }
        else if (! std::memcmp (d + i, "data", 4))
        {
            const bool pcm = tag == 1 && (bits == 16 || bits == 24), flt = tag == 3 && bits == 32;
            if (! (pcm || flt) || (ch != 1 && ch != 2) || rate < 8000 || rate > 384000) return false;
            const size_t bps = (size_t) bits / 8, frames = std::min<size_t> (body / (bps * ch), MAX_FRAMES);
            if (! frames) return false;
            out.l.assign (frames, 0.f);
            out.r.assign (frames, 0.f);
            out.rate = (float) rate;
            for (size_t f = 0; f < frames; ++f)
            {
                float v[2] = { 0.f, 0.f };
                for (int k = 0; k < ch; ++k)
                {
                    const uint8_t* s = c + (f * ch + k) * bps;
                    if (bits == 16) v[k] = (int16_t) u16 (s) / 32768.f;
                    else if (bits == 24) v[k] = ((int32_t) (s[0] << 8 | s[1] << 16 | (uint32_t) s[2] << 24) >> 8) / 8388608.f;
                    else { float x; std::memcpy (&x, s, 4); v[k] = std::isfinite (x) ? x : 0.f; }
                }
                out.l[f] = v[0];
                out.r[f] = ch == 2 ? v[1] : v[0];
            }
            return true;
        }
        i += 8 + (size_t) len + (len & 1);   // chunks are word-aligned
    }
    return false;
}
}
```

```cpp
// Voice.h: one drum voice, a sample player with linear interpolation. No Rack dependency.
// Author: Ari Klopfer / A_Development. MIT
#pragma once
#include "Sample.h"

struct Voice
{
    const Sample* smp = nullptr;   // owned by the module's pad slot; never freed while a voice plays it
    double pos = 0;
    float gain = 0;
    bool on = false;

    void trigger (const Sample* s, int vel)
    {
        if (! s || s->size() < 2) return;
        smp = s; pos = 0; gain = vel / 127.f; on = true;
    }

    void process (float sr, float& l, float& r)
    {
        l = r = 0.f;
        if (! on) return;
        const int i = (int) pos;
        if (i >= smp->size() - 1) { on = false; return; }
        const float f = (float) (pos - i);
        l = gain * (smp->l[i] + f * (smp->l[i + 1] - smp->l[i]));
        r = gain * (smp->r[i] + f * (smp->r[i + 1] - smp->r[i]));
        pos += smp->rate / sr;
    }
};
```

```cpp
// Sequencer.h: 16 patterns, the current one, the clock, and the hits of the step just fired. No Rack dependency.
// Author: Ari Klopfer / A_Development. MIT
#pragma once
#include "Pattern.h"
#include "StepClock.h"

struct Sequencer
{
    Pattern pat[PATTERNS];
    int cur = 0;
    StepClock clk;
    uint8_t hit[PADS] = {};   // velocity per pad for the step fire() last read; 0 = none

    Sequencer() { clk.length = pat[0].length; }

    void select (int i)
    {
        cur = std::max (0, std::min (PATTERNS - 1, i));
        clk.length = pat[cur].length;
    }
    void setLength (int n) { pat[cur].length = clk.length = std::max (1, std::min (MAXSTEPS, n)); }

    // call right after the clock fired a step; clk.step is always < MAXSTEPS
    void fire() { for (int p = 0; p < PADS; ++p) hit[p] = pat[cur].vel[p][clk.step]; }
};
```

```python
#!/usr/bin/env python3
"""gen_kit.py: synthesises the bundled starter kit into res/kit/*.wav (16-bit mono, 44.1 kHz). Fixed seed, stdlib only.
Run from anywhere: py -3.11 tools/gen_kit.py
Author: Ari Klopfer / A_Development. MIT
"""
import math
import os
import random
import struct
import wave

SR = 44100
ROOT = os.path.dirname(os.path.dirname(os.path.abspath(__file__)))
rng = random.Random(505)


def render(seconds, fn):
    return [fn(i / SR) for i in range(int(seconds * SR))]


def sweep(f0, f1, k, t):
    """sine whose pitch falls from f0 to f1 at rate k; phase is the integral of the frequency"""
    return math.sin(2 * math.pi * (f1 * t + (f0 - f1) * (1 - math.exp(-k * t)) / k))


def noise():
    return rng.uniform(-1, 1)


def hp(xs):
    """first difference: a crude high-pass that is enough for hats"""
    return [b - a for a, b in zip([0.0] + xs, xs)]


def kick():
    return render(0.45, lambda t: sweep(160, 48, 28, t) * math.exp(-t * 8))


def snare():
    return render(0.28, lambda t: 0.6 * noise() * math.exp(-t * 18) + 0.6 * math.sin(2 * math.pi * 190 * t) * math.exp(-t * 28))


def chat():
    return hp(render(0.09, lambda t: noise() * math.exp(-t * 55)))


def ohat():
    return hp(render(0.45, lambda t: noise() * math.exp(-t * 9)))


def clap():
    def f(t):
        burst = sum(math.exp(-((t - b) * 120)) for b in (0.0, 0.011, 0.023) if t >= b)
        return noise() * (0.7 * burst + 0.5 * math.exp(-t * 16))
    return render(0.32, f)


def rim():
    return render(0.06, lambda t: (0.7 * math.sin(2 * math.pi * 1700 * t) + 0.5 * noise()) * math.exp(-t * 80))


def tom():
    return render(0.45, lambda t: sweep(220, 110, 12, t) * math.exp(-t * 9))


def crash():
    return hp(render(1.4, lambda t: noise() * math.exp(-t * 3)))


def write(name, xs):
    peak = max(abs(x) for x in xs) or 1
    path = os.path.join(ROOT, 'res', 'kit', name + '.wav')
    os.makedirs(os.path.dirname(path), exist_ok=True)
    with wave.open(path, 'wb') as w:
        w.setnchannels(1)
        w.setsampwidth(2)
        w.setframerate(SR)
        w.writeframes(b''.join(struct.pack('<h', int(x / peak * 0.9 * 32767)) for x in xs))
    print('wrote', path, len(xs), 'frames')


if __name__ == '__main__':
    for name, fn in (('kick', kick), ('snare', snare), ('chat', chat), ('ohat', ohat), ('clap', clap), ('rim', rim), ('tom', tom), ('crash', crash)):
        write(name, fn())
```

```python
    """Parts must not overlap each other, their labels, the band, the screws or the panel edge. Must be empty."""
    bad = []

    for j, (a, ax, ay, ar) in enumerate(parts):

            bad.append('%s leaves the panel or enters the band' % a)
        for b, bx, by, br in parts[j + 1:]:
            if math.hypot(ax - bx, ay - by) < ar + br:
                bad.append('%s touches %s' % (a, b))
        for sx in (5.08, W - 10.16):
            for sy in (0.0, H - 5.08):
                if sx - ar < sx + 5.08 and sx < ax + ar and sy < ay + ar and ay - ar < sy + 5.08 and ax - ar < sx + 5.08:
                    bad.append('%s under a screw' % a)
    for n, k, x, y, lab in LAYOUT:   # a label must clear every other part
        w, base, size = width(lab, LABEL_SIZE[k]), label_base(k, y), LABEL_SIZE[k]
        box = (x - w / 2, base - 0.73 * size, x + w / 2, base)
        for m, mx, my, mr in parts:
            if m != n:
                nx, ny = min(max(mx, box[0]), box[2]), min(max(my, box[1]), box[3])
                if math.hypot(mx - nx, my - ny) < mr:
                    bad.append('%s label covers %s' % (n, m))
        if box[2] > W or box[3] > H:
            bad.append('%s label leaves the panel' % n)
```

```cpp
    float flash[PADS] = {};                   // pad light, 1 -> 0 after a hit
    FaceTap face;
    dsp::SchmittTrigger clockTrig, resetTrig;
    bool wasRun = false;

    Drums()
    {
        config (PARAMS_LEN, INPUTS_LEN, OUTPUTS_LEN, LIGHTS_LEN);
        configParam (BPM_PARAM, 30.f, 300.f, 120.f, "Tempo", " BPM");
        configParam (SWING_PARAM, 0.f, 0.5f, 0.f, "Swing", "%", 0.f, 200.f);
        configParam (LENGTH_PARAM, 1.f, 64.f, 16.f, "Steps in this pattern")->snapEnabled = true;
        configParam (PATTERN_PARAM, 0.f, 15.f, 0.f, "Pattern", "", 0.f, 1.f, 1.f)->snapEnabled = true;
        configSwitch (RUN_PARAM, 0.f, 1.f, 1.f, "Run", { "Stopped", "Running" });
        configParam (LEVEL_PARAM, 0.f, 2.f, 1.f, "Mix level", "%", 0.f, 100.f);
        configInput (CLOCK_INPUT, "Clock (one pulse per step)");
        configInput (RESET_INPUT, "Reset");
        configInput (RUN_INPUT, "Run gate");
        configOutput (OUT_L_OUTPUT, "Mix left");
        configOutput (OUT_R_OUTPUT, "Mix right");
        for (int p = 0; p < PADS; ++p)
        {
            audition[p] = false;
            configOutput (PAD_OUTPUT + p, "Pad " + std::to_string (p + 1));
        }
        loadStarterKit();
    }

    ~Drums()
    {
        for (int p = 0; p < PADS; ++p) { delete slot[p].smp; delete slot[p].incoming.load(); }
    }

    // ---- samples (UI thread) ----
    // An empty Sample means "silent pad"; the audio thread only ever sees whole samples.
    bool loadSample (int p, const std::string& path)
    {
        Sample* s = new Sample;
        bool ok = false;
        if (! path.empty())
        {
            try { std::vector<uint8_t> d = system::readFile (path); ok = wavfile::parse (d.data(), d.size(), *s); }
            catch (const std::exception&) {}
        }
        if (! ok) *s = Sample();
        slot[p].path = ok ? path : "";
        delete slot[p].incoming.exchange (s);   // a previous unadopted sample is dropped
        dirty = true;
        return ok;
    }
    void loadStarterKit()
    {
        static const char* names[8] = { "kick", "snare", "chat", "ohat", "clap", "rim", "tom", "crash" };
        for (int p = 0; p < 8; ++p) loadSample (p, asset::plugin (pluginInstance, std::string ("res/kit/") + names[p] + ".wav"));
    }
    std::string padName (int p) const { return slot[p].path.empty() ? "-" : system::getStem (slot[p].path); }

    // ---- audio thread ----
    void adopt()
    {
        for (int p = 0; p < PADS; ++p)
        {
            Sample* s = slot[p].incoming.exchange (nullptr);
            if (! s) continue;
            voice[p].on = false;   // never free a sample a voice is reading
            delete slot[p].smp;    // ponytail: frees on the audio thread; fine for a user-triggered load
            slot[p].smp = s;
        }
    }

    void hitPad (int p, int vel)
    {
        voice[p].trigger (slot[p].smp, vel);
        flash[p] = 1.f;
    }

    void process (const ProcessArgs& a) override
    {
        if (dirty.exchange (false)) adopt();

        const int pi = (int) std::round (params[PATTERN_PARAM].getValue());
        if (pi != seq.cur) { seq.select (pi); params[LENGTH_PARAM].setValue ((float) seq.pat[pi].length); }
        else seq.setLength ((int) std::round (params[LENGTH_PARAM].getValue()));
        seq.clk.bpm = params[BPM_PARAM].getValue();
        seq.clk.swing = params[SWING_PARAM].getValue();

        const bool run = params[RUN_PARAM].getValue() > 0.5f && (! inputs[RUN_INPUT].isConnected() || inputs[RUN_INPUT].getVoltage() > 1.f);
        if (run && ! wasRun) seq.clk.reset();
        wasRun = run;
        lights[RUN_LIGHT].setBrightness (run ? 1.f : 0.f);
        if (resetTrig.process (inputs[RESET_INPUT].getVoltage(), 0.1f, 2.f)) seq.clk.reset();

        bool fired = false;
        if (inputs[CLOCK_INPUT].isConnected())
        {
            if (clockTrig.process (inputs[CLOCK_INPUT].getVoltage(), 0.1f, 2.f) && run) fired = seq.clk.pulse();
        }
        else if (run) fired = seq.clk.tick (a.sampleTime);
        if (fired)
        {
            seq.fire();
            for (int p = 0; p < PADS; ++p) if (seq.hit[p]) hitPad (p, seq.hit[p]);
        }
        for (int p = 0; p < PADS; ++p) if (audition[p].exchange (false)) hitPad (p, 100);

        float l = 0.f, r = 0.f;
        for (int p = 0; p < PADS; ++p)
        {
            float vl, vr;
            voice[p].process (a.sampleRate, vl, vr);
            l += vl; r += vr;
            flash[p] = std::max (0.f, flash[p] - 6.f * a.sampleTime);
        }
        const float g = 5.f * params[LEVEL_PARAM].getValue();   // +-1 sample = +-5 V
        outputs[OUT_L_OUTPUT].setVoltage (l * g);
        outputs[OUT_R_OUTPUT].setVoltage (r * g);
        face.push (clamp ((l + r) * 0.5f, -1.f, 1.f), a.sampleRate);
    }

    // ---- patch file ----
    // Sample paths under the plugin folder are saved as "@plugin/..." so the patch survives a plugin move.
    std::string packPath (const std::string& p) const
    {
        const std::string root = asset::plugin (pluginInstance, "");
        return p.compare (0, root.size(), root) == 0 ? "@plugin/" + p.substr (root.size()) : p;
    }
    std::string unpackPath (const std::string& p) const
    {
        return p.compare (0, 8, "@plugin/") == 0 ? asset::plugin (pluginInstance, p.substr (8)) : p;
    }

    json_t* dataToJson() override
    {
        json_t* j = json_object();
        json_t* pats = json_array();
        for (int i = 0; i < PATTERNS; ++i)
        {
            json_t* o = json_object();
            json_object_set_new (o, "length", json_integer (seq.pat[i].length));
            json_t* rows = json_array();
            for (int p = 0; p < PADS; ++p) json_array_append_new (rows, json_string (rowToHex (seq.pat[i].vel[p]).c_str()));
            json_object_set_new (o, "rows", rows);
            json_array_append_new (pats, o);
        }
        json_object_set_new (j, "patterns", pats);
        json_t* smp = json_array();
        for (int p = 0; p < PADS; ++p) json_array_append_new (smp, json_string (packPath (slot[p].path).c_str()));
        json_object_set_new (j, "samples", smp);
        return j;
    }

    void dataFromJson (json_t* j) override
    {
        json_t* pats = json_object_get (j, "patterns");
        for (int i = 0; i < PATTERNS && json_is_array (pats) && i < (int) json_array_size (pats); ++i)
        {
            json_t* o = json_array_get (pats, i);
            if (json_t* len = json_object_get (o, "length"))
                if (json_is_integer (len)) seq.pat[i].length = (int) clamp ((int) json_integer_value (len), 1, MAXSTEPS);
            json_t* rows = json_object_get (o, "rows");
            for (int p = 0; p < PADS && json_is_array (rows) && p < (int) json_array_size (rows); ++p)
                if (json_t* row = json_array_get (rows, p))
                    if (json_is_string (row)) rowFromHex (json_string_value (row), seq.pat[i].vel[p]);
        }
        seq.clk.length = seq.pat[seq.cur].length;
        json_t* smp = json_object_get (j, "samples");
        for (int p = 0; p < PADS && json_is_array (smp) && p < (int) json_array_size (smp); ++p)
            if (json_t* s = json_array_get (smp, p))
                if (json_is_string (s)) loadSample (p, unpackPath (json_string_value (s)));
    }
};
```

```cpp
// 16 rows (pad 16 on top, pad 1 at the bottom) x 16 columns of the current page. Left click paints or erases, right click erases.
struct Roll : DragWidget
{
    Drums* m = nullptr;
    int paint = 0;   // value a drag writes: 0 erase, 100 write

    float gutter() const { return mm2px (6.f); }
    bool cellAt (Vec pos, int& pad, int& step) const
    {
        const float cw = (box.size.x - gutter()) / 16, ch = box.size.y / 16;
        if (pos.x < gutter() || pos.y < 0) return false;
        const int c = (int) ((pos.x - gutter()) / cw), row = (int) (pos.y / ch);
        if (c < 0 || c > 15 || row < 0 || row > 15) return false;
        pad = 15 - row;
        step = m->page.load() * 16 + c;
        return step < MAXSTEPS;
    }
    void apply (Vec pos)
    {
        int pad, step;
        if (! m || ! cellAt (pos, pad, step)) return;
        m->seq.pat[m->seq.cur].vel[pad][step] = (uint8_t) paint;
        m->sel = pad;
    }
    void onButton (const ButtonEvent& e) override
    {
        DragWidget::onButton (e);
        if (! m || e.action != GLFW_PRESS || (e.button != GLFW_MOUSE_BUTTON_LEFT && e.button != GLFW_MOUSE_BUTTON_RIGHT)) return;
        int pad, step;
        if (cellAt (e.pos, pad, step))
            paint = (e.button == GLFW_MOUSE_BUTTON_RIGHT || m->seq.pat[m->seq.cur].vel[pad][step]) ? 0 : 100;
        else if (e.pos.x < gutter() && e.pos.y >= 0)   // row number: select that pad and hear it
        {
            const int p = 15 - (int) (e.pos.y / (box.size.y / 16));
            if (p >= 0 && p < PADS) { m->sel = p; m->audition[p] = true; }
            return;
        }
        apply (e.pos);
    }
    void onDragMove (const DragMoveEvent& e) override { DragWidget::onDragMove (e); apply (dragPos); }

    void draw (const DrawArgs& a) override
    {
        namespace C = ui_colors;
        const float g = gutter(), cw = (box.size.x - g) / 16, ch = box.size.y / 16;
        const int pg = m ? m->page.load() : 0, sel = m ? m->sel.load() : 0;
        const Pattern* pt = m ? &m->seq.pat[m->seq.cur] : nullptr;
        const int play = m ? m->seq.clk.step : -1;
        for (int row = 0; row < 16; ++row)
        {
            const int pad = 15 - row;
            drawText (a, std::to_string (pad + 1), g / 2, row * ch + ch / 2, 9.f, pad == sel ? C::magenta() : C::dim());
            for (int c = 0; c < 16; ++c)
            {
                const int s = pg * 16 + c;
                const bool inLen = ! pt || s < pt->length;
                nvgBeginPath (a.vg);
                nvgRect (a.vg, g + c * cw + 0.5f, row * ch + 0.5f, cw - 1.f, ch - 1.f);
                nvgFillColor (a.vg, ! inLen ? C::cellOff() : ((c / 4) & 1) ? C::cellBeat() : C::cell());
                nvgFill (a.vg);
                if (pt && pt->vel[pad][s])
                {
                    nvgFillColor (a.vg, C::lime (90 + pt->vel[pad][s] * 165 / 127));
                    nvgFill (a.vg);
                }

                {
                    nvgStrokeColor (a.vg, C::magenta());
                    nvgStrokeWidth (a.vg, 1.f);
                    nvgStroke (a.vg);
                }
            }
        }
    }
```

```cpp
// Adsr.h: a linear gate ADSR with a fast choke. No Rack dependency.
// a, d, r are seconds; s is a level 0..1. gate(true) retriggers from the current level (no click).
// Author: Ari Klopfer / A_Development. MIT
#pragma once
#include <algorithm>

struct Adsr
{
    enum State { IDLE, ATK, DEC, SUS, REL };
    State st = IDLE;
    float v = 0.f, a = 0.001f, d = 0.2f, s = 1.f, r = 0.05f, rate = 0.f;

    bool active() const { return st != IDLE; }

    void gate (bool on)
    {
        if (on) st = ATK;
        else if (st != IDLE) { st = REL; rate = v / std::max (r, 1e-4f); }
    }
    // ~5 ms fade whatever the release knob says (choke groups)
    void choke()
    {
        if (st == IDLE) return;
        st = REL;
        rate = v / 0.005f;
    }

    float next (float dt)
    {
        switch (st)
        {
            case ATK: v += dt / std::max (a, 1e-4f); if (v >= 1.f) { v = 1.f; st = DEC; } break;
            case DEC: v -= (1.f - s) * dt / std::max (d, 1e-4f); if (v <= s) { v = s; st = SUS; } break;
            case SUS: v = s; break;
            case REL: v -= rate * dt; if (v <= 0.f) { v = 0.f; st = IDLE; } break;
            case IDLE: break;
        }
        return v;
    }
};
```

```cpp
// Filter.h: stereo one-pole low-pass then high-pass for a pad. No Rack dependency.
// Author: Ari Klopfer / A_Development. MIT
#pragma once
#include <algorithm>
#include <cmath>

struct OnePole
{
    float z = 0.f;
    float process (float x, float a) { z += a * (x - z); return z; }
};

inline float onePoleCoef (float fc, float sr)
{
    fc = std::max (1.f, std::min (fc, 0.45f * sr));
    return 1.f - std::exp (-6.2831853f * fc / sr);
}

struct PadFilter
{
    OnePole lpL, lpR, hpL, hpR;
    float lpHz = 20000.f, hpHz = 20.f;

    void process (float& l, float& r, float sr)
    {
        const float al = onePoleCoef (lpHz, sr), ah = onePoleCoef (hpHz, sr);
        l = lpL.process (l, al);
        r = lpR.process (r, al);
        l -= hpL.process (l, ah);   // high-pass = input minus its own low-pass
        r -= hpR.process (r, ah);
    }
};
```

```cpp
// Comp.h: feed-forward peak compressor for a pad. No Rack dependency.
// gain() returns the linear gain to multiply both channels by (reduction and makeup in one).
// Author: Ari Klopfer / A_Development. MIT
#pragma once
#include <algorithm>
#include <cmath>

struct Comp
{
    float thr = 0.f, ratio = 1.f, atk = 5.f, rel = 80.f, makeup = 0.f;   // dB, :1, ms, ms, dB
    float env = 0.f;

    float gain (float inL, float inR, float sr)
    {
        const float x = std::max (std::fabs (inL), std::fabs (inR));
        const float ms = x > env ? atk : rel;
        env += (1.f - std::exp (-1.f / (std::max (ms, 0.01f) * 0.001f * sr))) * (x - env);
        const float over = 20.f * std::log10 (std::max (env, 1e-6f)) - thr;
        const float red = (over > 0.f && ratio > 1.f) ? over * (1.f - 1.f / ratio) : 0.f;
        return std::pow (10.f, (makeup - red) / 20.f);
    }
};
```

```cpp
// Voice.h: one drum voice. Sample player -> amp ADSR -> low/high-pass -> compressor -> volume/pan.
// PadCfg is the per-pad sound setting. No Rack dependency.
// Author: Ari Klopfer / A_Development. MIT
#pragma once
#include "Adsr.h"
#include "Comp.h"
#include "Filter.h"
#include "Pattern.h"
#include "Sample.h"

struct PadCfg
{
    float vol = 1.f, pan = 0.f, pitch = 0.f;                     // gain, -1..1, semitones
    float atk = 0.001f, dec = 0.2f, sus = 1.f, rel = 0.05f;      // seconds, level
    float lp = 20000.f, hp = 20.f;                               // Hz
    float thr = 0.f, ratio = 1.f, catk = 5.f, crel = 80.f, makeup = 0.f;   // dB, :1, ms, ms, dB
    bool gateMode = false;                                       // false: one-shot, true: release after one step
    int choke = 0;                                               // 0 = none; pads sharing a number cut each other
};

struct Voice
{
    const Sample* smp = nullptr;   // owned by the module's pad slot; never freed while a voice plays it
    double pos = 0, gateLeft = 0;
    float vel = 0.f;
    bool on = false, released = false;
    Adsr env;
    PadFilter filt;
    Comp comp;

    void trigger (const Sample* s, int v, const PadCfg& c, double stepSec)
    {
        if (! s || s->size() < 2) return;
        smp = s; pos = 0; vel = v / 127.f; on = true; released = false;
        gateLeft = stepSec;
        (void) c;
        env.gate (true);
    }
    void choke() { env.choke(); }

    void process (const PadCfg& c, float sr, float& l, float& r)
    {
        l = r = 0.f;
        if (! on) return;
        env.a = c.atk; env.d = c.dec; env.s = c.sus; env.r = c.rel;
        const double rate = smp->rate / sr * std::pow (2.0, c.pitch / 12.0);   // sample frames per output frame
        const int i = (int) pos;
        if (i >= smp->size() - 1) { on = false; return; }

        // One-shot: start releasing so the tail ends with the sample. Gate: release after one step.
        if (! released)
        {
            const bool end = c.gateMode ? (gateLeft -= 1.0 / sr) <= 0.0 : (smp->size() - 1 - pos) / (rate * sr) <= c.rel;
            if (end) { env.gate (false); released = true; }
        }
        const float f = (float) (pos - i);
        l = smp->l[i] + f * (smp->l[i + 1] - smp->l[i]);
        r = smp->r[i] + f * (smp->r[i + 1] - smp->r[i]);
        pos += rate;

        const float e = env.next (1.f / sr) * vel;
        if (! env.active()) { on = false; l = r = 0.f; return; }
        l *= e; r *= e;

        filt.lpHz = c.lp; filt.hpHz = c.hp;
        filt.process (l, r, sr);
        comp.thr = c.thr; comp.ratio = c.ratio; comp.atk = c.catk; comp.rel = c.crel; comp.makeup = c.makeup;
        const float g = comp.gain (l, r, sr) * c.vol;
        l *= g * (c.pan > 0.f ? 1.f - c.pan : 1.f);
        r *= g * (c.pan < 0.f ? 1.f + c.pan : 1.f);
    }
};

// A pad fired: cut every other voice in its choke group (open hat by closed hat).

{

    for (int q = 0; q < PADS; ++q)
        if (q != pad && c[q].choke == c[pad].choke) v[q].choke();
}
```

```cpp
// Beats.h: factory beats as text. 16 characters per row: 'x' hit (100), 'o' soft hit (60), '.' nothing.
// Pad map: 0 kick, 1 snare, 2 closed hat, 3 open hat, 4 clap, 5 rim, 6 tom, 7 crash. No Rack dependency.
// Author: Ari Klopfer / A_Development. MIT
#pragma once
#include "Pattern.h"

struct BeatRow { int pad; const char* steps; };
struct Beat { const char* name; BeatRow rows[4]; };   // pad < 0 ends a short list

constexpr int NUM_BEATS = 3;
static const Beat BEATS[NUM_BEATS] = {
    { "Boom bap", { { 0, "x.....x..x......" }, { 1, "....x.......x..." }, { 2, "x.x.x.x.x.x.x.x." }, { -1, "" } } },
    { "House",    { { 0, "x...x...x...x..." }, { 4, "....x.......x..." }, { 3, "..x...x...x...x." }, { 2, "o.o.o.o.o.o.o.o." } } },
    { "Trap",     { { 0, "x..x......x....." }, { 4, "........x......." }, { 2, "xoxoxoxoxoxoxxxo" }, { 3, "..............x." } } },
};

inline void applyBeat (Pattern& p, const Beat& b)
{
    p.clear();
    p.length = 16;
    for (const BeatRow& r : b.rows)
    {
        if (r.pad < 0) break;
        for (int s = 0; s < 16 && r.steps[s]; ++s) p.vel[r.pad][s] = r.steps[s] == 'x' ? 100 : r.steps[s] == 'o' ? 60 : 0;
    }
}
```
